sudo mount -t devpts none ~/netra/root/dev/pts

///Masuk ke lingkungan Chroot
sudo chroot ~/netra/root /bin/bash

Window.SetBackgroundTopColor(0.02, 0.05, 0.09);
Window.SetBackgroundBottomColor(0.02, 0.05, 0.09);

logo_image = Image("boot.png").Scale(300, 300);

logo_sprite = Sprite(logo_image);

logo_sprite.SetX((Window.GetWidth() - logo_image.GetWidth()) / 2);
logo_sprite.SetY((Window.GetHeight() - logo_image.GetHeight()) / 2);
logo_sprite.SetOpacity(1);


message_sprite = Sprite();

msg_x = (Window.GetWidth() - 400) / 2;
msg_y = (Window.GetHeight() / 2) + 160;

message_sprite.SetPosition(msg_x, msg_y, 10000);

fun message_callback (text) {
    my_image = Image.Text(text, 0.8, 0.9, 1.0, 1.0);
    message_sprite.SetImage(my_image);
}

Plymouth.SetMessageFunction(message_callback);


total_frames = 16;
wm_width = 300;
wm_height = 100;

for (i = 0; i < total_frames; i++) {
    wm_images[i] = Image((i + 1) + ".png").Scale(wm_width, wm_height);
}

wm_sprite = Sprite();
wm_sprite.SetImage(wm_images[0]);
wm_sprite.SetX((Window.GetWidth() - wm_width) / 2);
wm_sprite.SetY(Window.GetHeight() - wm_height - 80);
wm_sprite.SetOpacity(1);

wm_frame = 0;
wm_tick = 0;

fun refresh_callback () {
    wm_tick++;
    if (wm_tick % 3 == 0) {
        wm_sprite.SetImage(wm_images[wm_frame]);
        wm_frame = (wm_frame + 1) % total_frames;
    }
}

Plymouth.SetRefreshFunction(refresh_callback);

===== BUILD & CETAK ISO NETRA OS =====

17. Lepas Mount Sistem Virtual (Di Terminal Host)
sudo umount -lf ~/netra/root/dev/pts
sudo umount -lf ~/netra/root/dev
sudo umount -lf ~/netra/root/run
sudo umount -lf ~/netra/root/proc
sudo umount -lf ~/netra/root/sys

18. Perbarui Kernel & Initrd di Folder ISO (Wajib agar Plymouth Custom Muncul)
/// Pastikan nama file vmlinuz dan initrd.img sesuai dengan versi di ~/netra/root/boot/
sudo cp ~/netra/root/boot/vmlinuz-7.0.0-43-generic ~/netra/iso/casper/vmlinuz
sudo cp ~/netra/root/boot/initrd.img-7.0.0-43-generic ~/netra/iso/casper/initrd

19. Kemas Ulang SquashFS & Ukuran Filesystem
/// Hapus file squashfs & size lama agar tidak bentrok dengan sistem gabungan
sudo rm -f ~/netra/iso/casper/*.squashfs ~/netra/iso/casper/*.size

/// Buat ulang file squashfs dari sistem gabungan (~/netra/root)
sudo mksquashfs ~/netra/root ~/netra/iso/casper/minimal.standard.live.squashfs -comp zstd -Xcompression-level 19 -noappend

/// Hitung ulang ukuran filesystem untuk installer
printf $(sudo du -sx --block-size=1 ~/netra/root | cut -f1) | sudo tee ~/netra/iso/casper/minimal.standard.live.size

/// Lepas mount OverlayFS
sudo umount ~/netra/root

20. Ubah Metadata Disk (Seperti di Cubic: Disk Name, Release, Release URL)
echo 'Netra OS 1.0 LTS amd64 "Kelompok 6" - Release amd64' > ~/netra/iso/.disk/info
echo 'https://netra-os.vercel.app/' > ~/netra/iso/.disk/release_notes_url

21. Perbarui Checksum MD5
cd ~/netra/iso
rm -f md5sum.txt
find -type f -print0 | xargs -0 md5sum | grep -v "./boot/" | grep -v "./md5sum.txt" > md5sum.txt

22. Cek Parameter Boot & Cetak File ISO (netra-os-1.0-desktop-amd64.iso)
cd ~/netra

/// Cek angka sektor El Torito dari ISO asli (cocokkan dengan perintah xorriso di bawah)
xorriso -indev ubuntu-26.04.1-desktop-amd64.iso -report_el_torito as_mkisofs

/// Cetak ISO Final
xorriso -as mkisofs -iso-level 3 -r -V 'Netra OS 1.0 LTS amd64' -o ~/netra/netra-os-1.0-desktop-amd64.iso --grub2-mbr --interval:local_fs:0s-15s:zero_mbrpt,zero_gpt:'ubuntu-26.04.1-desktop-amd64.iso' --protective-msdos-label -partition_cyl_align off -partition_offset 16 --mbr-force-bootable -append_partition 2 28732ac11ff8d211ba4b00a0c93ec93b --interval:local_fs:12649996d-12660291d::'ubuntu-26.04.1-desktop-amd64.iso' -appended_part_as_gpt -iso_mbr_part_type a2a0d0ebe5b9334487c068b6b72699c7 -c '/boot.catalog' -b '/boot/grub/i386-pc/eltorito.img' -no-emul-boot -boot-load-size 4 -boot-info-table --grub2-boot-info -eltorito-alt-boot -e '--interval:appended_partition_2_start_3162499s_size_10296d:all::' -no-emul-boot -boot-load-size 10296 ~/netra/iso

18. Perbarui Kernel & Initrd di Folder ISO (Wajib agar Plymouth Custom Muncul)

19. Kemas Ulang SquashFS & Ukuran Filesystem
/// Hapus file squashfs & size lama agar tidak bentrok dengan sistem gabungan
sudo rm -f ~/netra/iso/casper/*.squashfs ~/netra/iso/casper/*.size

/// Buat ulang file squashfs dari sistem gabungan (~/netra/root)
sudo mksquashfs ~/netra/root ~/netra/iso/casper/minimal.standard.live.squashfs -comp zstd -Xcompression-level 19 -noappend

/// Hitung ulang ukuran filesystem untuk installer
printf $(sudo du -sx --block-size=1 ~/netra/root | cut -f1) | sudo tee ~/netra/iso/casper/minimal.standard.live.size

/// Lepas mount OverlayFS
sudo umount ~/netra/root
