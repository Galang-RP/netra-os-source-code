nano /usr/share/plymouth/themes/netra-logo/netra-logo.script

Window.SetBackgroundTopColor(0.02, 0.05, 0.09);
Window.SetBackgroundBottomColor(0.02, 0.05, 0.09);

logo_image = Image("boot.png").Scale(300, 300);

logo_sprite = Sprite(logo_image);

logo_sprite.SetX((Window.GetWidth() - logo_image.GetWidth()) / 2);
logo_sprite.SetY((Window.GetHeight() - logo_image.GetHeight()) / 2);
logo_sprite.SetOpacity(1);


message_sprite = Sprite();


msg_x = (Window.GetWidth() - 400) / 2;
msg_y = (Window.GetHeight() / 2) + 160;

message_sprite.SetPosition(msg_x, msg_y, 10000);

fun message_callback (text) {
    my_image = Image.Text(text, 0.8, 0.9, 1.0, 1.0);
    message_sprite.SetImage(my_image);
}

Plymouth.SetMessageFunction(message_callback);

total_frames = 16;
wm_width = 300;
wm_height = 100;

for (i = 0; i < total_frames; i++) {
    wm_images[i] = Image((i + 1) + ".png").Scale(wm_width, wm_height);
}

wm_sprite = Sprite();
wm_sprite.SetImage(wm_images[0]);
wm_sprite.SetX((Window.GetWidth() - wm_width) / 2);
wm_sprite.SetY(Window.GetHeight() - wm_height - 80);
wm_sprite.SetOpacity(1);

wm_frame = 0;
wm_tick = 0;

fun refresh_callback () {
    wm_tick++;
    if (wm_tick % 3 == 0) {
        wm_sprite.SetImage(wm_images[wm_frame]);
        wm_frame = (wm_frame + 1) % total_frames;
    }
}

Plymouth.SetRefreshFunction(refresh_callback);




update-alternatives --install \
/usr/share/plymouth/themes/default.plymouth \
default.plymouth \
/usr/share/plymouth/themes/netra-logo/netra-logo.plymouth \
200


update-alternatives --set default.plymouth \
/usr/share/plymouth/themes/netra-logo/netra-logo.plymouth


readlink -f /usr/share/plymouth/themes/default.plymouth
update-alternatives --query default.plymouth

11. //skrip web
nano /usr/local/bin/netra-welcome.sh

#!/bin/bash

while pgrep -x "gnome-initial-setup" > /dev/null || pgrep -x "oem-config" > /dev/null || pgrep -f "subiquity" > /dev/null; do
    sleep 4
done

MARKER="$HOME/.config/netra-welcome-done"

if [ -f "$MARKER" ]; then
    exit 0
fi

sleep 10

# Buka halaman GitHub Netra OS
if xdg-open "https://netra-os.vercel.app" >/dev/null 2>&1; then
    mkdir -p "$HOME/.config"
    touch "$MARKER"
fi

exit 0



chmod +x /usr/local/bin/netra-welcome.sh
mkdir -p /etc/xdg/autostart


nano /etc/xdg/autostart/netra-welcome.desktop

[Desktop Entry]
