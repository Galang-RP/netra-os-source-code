# ==========================================
# BAGIAN 1: PERSIAPAN & MASUK CHROOT
# ==========================================

# 1. Bersihkan sisa folder kerja lama
sudo umount -lf ~/netra/root/dev/pts ~/netra/root/dev ~/netra/root/run ~/netra/root/proc ~/netra/root/sys ~/netra/root 2>/dev/null
rm -rf ~/netra/iso ~/netra/root ~/netra/netra-os1.0-desktop-amd64.iso

# 2. Install tools remastering
sudo apt update && sudo apt install -y squashfs-tools xorriso

# 3. Buat struktur direktori kerja (menggunakan folder 'iso')
mkdir -p ~/netra/iso ~/netra/root

# 4. Ekstrak seluruh isi ISO asli ke folder iso
cd ~/netra
xorriso -osirrox on -indev ubuntu-26.04.1-desktop-amd64.iso -extract / ~/netra/iso
chmod -R u+w ~/netra/iso

# 5. Ekstrak file inti sistem (minimal.squashfs) langsung ke folder root
sudo unsquashfs -f -d ~/netra/root ~/netra/iso/casper/minimal.squashfs

# 6. Mount sistem virtual untuk chroot
sudo mount --bind /dev ~/netra/root/dev
sudo mount --bind /dev/pts ~/netra/root/dev/pts
sudo mount -t proc proc ~/netra/root/proc
sudo mount -t sysfs sys ~/netra/root/sys
sudo mount -t tmpfs tmpfs ~/netra/root/run

# 7. Salin konfigurasi DNS dengan aman
sudo rm -f ~/netra/root/etc/resolv.conf
sudo cp /etc/resolv.conf ~/netra/root/etc/resolv.conf

# 8. Masuk ke lingkungan Chroot
sudo chroot ~/netra/root /bin/bash

# ==========================================
# BAGIAN 3: BUILD & CETAK ISO (SETELAH KELUAR CHROOT)
# ==========================================

# 1. Lepas mount sistem virtual
sudo umount -lf ~/netra/root/dev/pts ~/netra/root/dev ~/netra/root/run ~/netra/root/proc ~/netra/root/sys 2>/dev/null

# 2. Kemas Ulang SquashFS & Update minimal.size (mengarah ke folder 'iso')
sudo rm -f ~/netra/iso/casper/minimal.squashfs ~/netra/iso/casper/minimal.size

sudo mksquashfs ~/netra/root ~/netra/iso/casper/minimal.squashfs -comp zstd -Xcompression-level 15 -noappend

# Hitung ukuran filesystem akurat pakai sudo
printf $(sudo du -s --bytes ~/netra/root | cut -f1) | sudo tee ~/netra/iso/casper/minimal.size

# 3. Ubah Metadata Disk (mengarah ke folder 'iso')
echo 'Netra OS 1.0 LTS amd64 "Kelompok 6"' > ~/netra/iso/.disk/info
echo 'https://netra-os.vercel.app/' > ~/netra/iso/.disk/release_notes_url

# 4. Perbarui Checksum MD5 (mengarah ke folder 'iso')
cd ~/netra/iso
rm -f md5sum.txt
find -type f -print0 | xargs -0 md5sum | grep -v "./boot/" | grep -v "./md5sum.txt" > md5sum.txt

# 5. Cetak File ISO Final (menggunakan folder 'iso' sebagai sumber)
cd ~/netra
sudo xorriso -as mkisofs \
  -iso-level 3 \
  -full-iso9660-filenames \
  -volid "Netra OS 1.0 LTS amd64" \
  -eltorito-boot boot/grub/i386-pc/eltorito.img \
  -eltorito-catalog boot/grub/i386-pc/boot.catalog \
  -no-emul-boot -boot-load-size 4 -boot-info-table \
  -eltorito-alt-boot \
  -e EFI/boot/bootx64.efi \
  -no-emul-boot -isohybrid-gpt-basdat \
  -output ~/netra/netra-os1.0-desktop-amd64.iso \
  ~/netra/iso
